/* ===== LE MOTEUR BÛCHE CYBER (2026-11-09, Léo : « le concept du moteur bûche actuel, remixé par tous les trucs bien de ce moteur [le cyber] » ;
   « le boost on le laisse tel quel ») =====
   ?moteur=buche : la BÛCHE (voix-moteur.js, graphe-audio.js) garde TOUT ; on lui ajoute, calé sur sa note (assets/audio/moteur-buche/moteur-buche.js) :
     · la MÉCANIQUE du moteur vivant, branchée DANS la bûche (engMix) : ses coups passent par la saturation, la gorge et le filtre du jeu ;
     · le CYBER (cœur light-cycle, onduleur à marches, plasma, clic de rapport), branché après le filtre, au volume de la bûche (`ev`) ;
     · L'ÉTAGE DE PUISSANCE sur la somme : le niveau suit la charge (pousser +2/+3 dB, lâcher −2/−3 dB ; le boost n'y entre pas), deux coupes
       anti-fatigue (−4 dB à 3 kHz, −4 dB au-dessus de 5,5 kHz), le creux de 60 ms au passage de rapport.
   Sans le drapeau : rien n'est chargé (MB.on faux, chaque appel sort tout de suite). `dbgMB()` à la console. */
const MB={on:/[?&]moteur=buche\b/.test(location.search),ok:false,node:null,mG:null,cG:null,puis:null,creux:null,eq1:null,eq2:null,msgT:0,vu:0,KM:8,KC:4};
MB.div=function(lv){const e=ENG_SND[lv]||ENG_SND[0];return Math.max(1,Math.pow(2,Math.ceil(Math.log2(engGenou((e.i+e.r)*.62)/240))));};
MB.envoi=function(o){if(MB.node)try{MB.node.port.postMessage(o);}catch(e){}};
MB.build=function(){
  if(!MB.on||MB.node||!AC||!AC.audioWorklet||!gainN)return;
  const A='assets/audio/',v='?v=1',m=function(f){return function(){return AC.audioWorklet.addModule(A+f);};};
  AC.audioWorklet.addModule(A+'moteur-vivant/moteur-vivant.js?v=2').then(m('moteur-cyber/moteur-cyber.js'+v)).then(m('moteur-buche/moteur-buche.js'+v)).then(m('moteur-buche/worklet-buche.js'+v)).then(function(){
    const lv=engLvl<0?0:engLvl,n=new AudioWorkletNode(AC,'moteur-buche',{numberOfInputs:0,numberOfOutputs:2,outputChannelCount:[2,2],processorOptions:{palier:lv,div:MB.div(lv)}});
    const mG=AC.createGain();mG.gain.value=0;n.connect(mG,0);mG.connect(engMix);
    const cG=AC.createGain();cG.gain.value=0;n.connect(cG,1);
    const puis=AC.createGain(),creux=AC.createGain(),eq1=AC.createBiquadFilter(),eq2=AC.createBiquadFilter(),hp=AC.createBiquadFilter();
    eq1.type='peaking';eq1.frequency.value=3000;eq1.Q.value=.8;eq1.gain.value=-4;eq2.type='highshelf';eq2.frequency.value=5500;eq2.gain.value=-4;
    hp.type='highpass';hp.frequency.value=28;
    gainN.disconnect(MASTER);gainN.connect(puis);cG.connect(puis);puis.connect(creux);creux.connect(eq1);eq1.connect(eq2);eq2.connect(hp);hp.connect(MASTER);
    Object.assign(MB,{node:n,mG:mG,cG:cG,puis:puis,creux:creux,eq1:eq1,eq2:eq2,ok:true});
    /* le garde-fou : plus de commande depuis 0,7 s (menu, pause, garage) → le cyber et la mécanique se taisent, la bûche reste seule et nue */
    setInterval(function(){if(MB.ok&&performance.now()-MB.vu>700)MB.repos();},250);
  }).catch(function(e){MB.on=false;try{console.warn('moteur bûche cyber : la bûche seule',e);}catch(_){}});
};
MB.tier=function(lv){if(MB.ok)MB.envoi({palier:lv,div:MB.div(lv)});};
MB.rapport=function(){if(!MB.ok)return;MB.envoi({ev:'rapport'});const T=AC.currentTime,g=MB.creux.gain;g.cancelScheduledValues(T);g.setTargetAtTime(.6,T,.006);g.setTargetAtTime(1,T+.06,.03);};
MB.drive=function(dt,P9,rpm,thr,charge){
  if(!MB.ok)return;MB.vu=performance.now();const T=AC.currentTime;
  stt(MB.puis.gain,.80+.45*Math.min(1,charge)+.12*thr,T,.06);stt(MB.cG.gain,P9.ev*MB.KC,T,.08);stt(MB.mG.gain,MB.KM,T,.05);
  MB.msgT+=dt;if(MB.msgT>.025){MB.msgT=0;MB.envoi({c:{hz:P9.fF,rpmN:Math.min(1.02,rpm),thr:thr,load:thr?Math.min(1,charge):0,boost:nA}});}
};
MB.repos=function(){if(!MB.ok)return;const T=AC.currentTime;stt(MB.cG.gain,0,T,.06);stt(MB.mG.gain,0,T,.06);stt(MB.puis.gain,1,T,.1);MB.envoi({c:{thr:0,load:0}});};
window.dbgMB=function(){return {on:MB.on,ok:MB.ok,puis:MB.puis?+MB.puis.gain.value.toFixed(3):null,cyber:MB.cG?+MB.cG.gain.value.toFixed(4):null,meca:MB.mG?+MB.mG.gain.value.toFixed(2):null};};
