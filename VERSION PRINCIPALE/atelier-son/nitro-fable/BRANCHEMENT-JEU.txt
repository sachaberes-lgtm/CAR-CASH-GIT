/* ===== LE NITRO PHYSIQUE (2026-11-10) : une flamme modélisée dans le tuyau + une mécanique douce, calées sur la note du moteur bûche cyber (assets/audio/moteur-vivant/nitro-physique.js).
   ?nitro=physique : sans le drapeau, rien n'est chargé (NX.on faux, chaque appel sort tout de suite). Il se branche à l'entrée du creux de l'étage de puissance de MB. */
const NX={on:/[?&]nitro=physique\b/.test(location.search),ok:false,node:null,g:null,msgT:0,preset:'forge',niv:.11};
NX.envoi=function(o){if(NX.node)try{NX.node.port.postMessage(o);}catch(e){}};
NX.build=function(){
  if(!NX.on||NX.node||!AC||!AC.audioWorklet||!MB.ok)return;
  const A='assets/audio/moteur-vivant/',v='?v=1',m=function(f){return function(){return AC.audioWorklet.addModule(A+f+v);};};
  m('nitro-matiere.js')().then(m('nitro-physique.js')).then(m('worklet-nitro-physique.js')).then(function(){
    const n=new AudioWorkletNode(AC,'nitro-physique',{numberOfInputs:0,numberOfOutputs:1,outputChannelCount:[2],processorOptions:{preset:NX.preset}});
    const g=AC.createGain();g.gain.value=NX.niv;n.connect(g);g.connect(MB.creux);
    Object.assign(NX,{node:n,g:g,ok:true});
  }).catch(function(e){NX.on=false;try{console.warn('nitro physique : sans',e);}catch(_){}});
};
NX.drive=function(dt,hz,thr,charge,nA,vit){if(!NX.ok)return;NX.msgT+=dt;if(NX.msgT>.025){NX.msgT=0;NX.envoi({c:{f0:hz,thr:thr,load:thr?Math.min(1,charge):0,nit:Math.min(1,nA),speed:Math.min(1,vit)}});}};
NX.rapport=function(){NX.envoi({ev:'rapport'});};
NX.repos=function(){if(!NX.ok)return;NX.envoi({c:{nit:0,thr:0,load:0}});};
